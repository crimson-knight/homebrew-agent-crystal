class AcceptancePerson0000 < Grant::Base

  column email : String
  column age : Int32
  column is_active : Bool
  column score : Float64
  column country_code : String
  column notes : String?
  column rating : Int32?
  column balance : Float64?
  column group_number : Int64
  column settings_text : String?
  column created_at : Time
